-- Egg Hatcher v2 | executor script
-- Hatches the nearest egg (optionally only eggs whose name matches a filter, default "Golden Grove").
-- RightShift shows/hides the window. "x" stops everything and cleans up.
--
-- Speed engine: it never pauses. It keeps many requests in flight and spaces them with a pace
-- that tunes itself to the server's limit: a rejected request widens the spacing a little, a run of
-- accepted ones tightens it again, so it keeps probing for the fastest pace the server accepts.
-- Each request always asks for the biggest hatch amount you can afford.

local FILE = "EggHatcher_Stats.json"
local cfg = { filter = "Golden Grove", range = 60, skipAnim = true }
local stats = { total = 0, perEgg = {} }
local statsDirty = false

local function loadSaved()
	if type(isfile) ~= "function" or type(readfile) ~= "function" then
		return
	end
	local ok, data = pcall(function()
		if not isfile(FILE) then
			return nil
		end
		return HttpService:JSONDecode(readfile(FILE))
	end)
	if ok and type(data) == "table" then
		if type(data.stats) == "table" then
			stats.total = tonumber(data.stats.total) or 0
			if type(data.stats.perEgg) == "table" then
				stats.perEgg = data.stats.perEgg
			end
		end
		if type(data.cfg) == "table" then
			if type(data.cfg.filter) == "string" then cfg.filter = data.cfg.filter end
			if tonumber(data.cfg.range) then cfg.range = math.clamp(tonumber(data.cfg.range), 5, 500) end
			if type(data.cfg.skipAnim) == "boolean" then cfg.skipAnim = data.cfg.skipAnim end
		end
	end
end

local function saveNow()
	if type(writefile) ~= "function" then
		return
	end
	pcall(function()
		writefile(FILE, HttpService:JSONEncode({ stats = stats, cfg = cfg }))
	end)
	statsDirty = false
end

local canSetIdentity = type(setthreadidentity) == "function"
local IDENTITY_EXEC = 7
if type(getthreadidentity) == "function" then
	local ok, id = pcall(getthreadidentity)
	if ok and type(id) == "number" then
		IDENTITY_EXEC = id
	end
end

-- game modules expect a game-level thread identity, so every call goes through here
local function gcall(fn, ...)
	if type(fn) ~= "function" then
		return false, "not a function"
	end
	if canSetIdentity then
		pcall(setthreadidentity, 2)
	end
	local r = table.pack(pcall(fn, ...))
	if canSetIdentity then
		pcall(setthreadidentity, IDENTITY_EXEC)
	end
	return table.unpack(r, 1, r.n)
end

local function req(path)
	local ok, mod = gcall(function()
		local node = RS
		for seg in string.gmatch(path, "[^%.]+") do
			node = node:FindFirstChild(seg)
			if not node then
				error("missing " .. path)
			end
		end
		return require(node)
	end)
	if ok then
		return mod
	end
	return nil
end

---------------------------------------------------------------- egg picking
local function allEggs()
	local out = {}
	if not (G.CustomEggs and type(G.CustomEggs.All) == "function") then
		return out
	end
	local ok, all = gcall(G.CustomEggs.All)
	if not ok or type(all) ~= "table" then
		return out
	end
	for uid, e in pairs(all) do
		if type(e) == "table" then
			local pos = e._position
			if typeof(pos) ~= "Vector3" and e._model and e._model.Parent then
				pcall(function()
					pos = e._model:GetPivot().Position
				end)
			end
			if typeof(pos) == "Vector3" then
				out[#out + 1] = {
					uid = e._uid or uid,
					id = tostring(e._id or "?"),
					dir = e._dir or (G.Eggs and G.Eggs[e._id]) or nil,
					pos = pos,
					hatchable = e._hatchable,
					maxCount = e._maxEggCount,
				}
			end
		end
	end
	return out
end

local function matchesFilter(id)
	local f = string.lower(cfg.filter or "")
	if f == "" then
		return true
	end
	return string.find(string.lower(id), f, 1, true) ~= nil
end

local function maxHatch(egg)
	if egg.maxCount and type(egg.maxCount) == "number" and egg.maxCount >= 1 then
		return math.floor(egg.maxCount)
	end
	if G.EggCmds and type(G.EggCmds.GetMaxHatch) == "function" then
		local ok, n = gcall(G.EggCmds.GetMaxHatch, egg.dir)
		if ok and type(n) == "number" and n >= 1 then
			return math.floor(n)
		end
	end
	return 1
end

local regularOffUntil = 0
local target, targetAt, targetNote = nil, 0, ""

-- nearest matching custom egg in range; falls back to a regular egg named by the filter (experimental)
local function refreshTarget()
	targetAt = os.clock()
	local root = hrp()
	if not root then
		target, targetNote = nil, "no character"
		return
	end
	local p = root.Position
	local best, bestD
	for _, e in ipairs(allEggs()) do
		if matchesFilter(e.id) and e.hatchable ~= false then
			local d = (e.pos - p).Magnitude
			if d <= cfg.range and (not bestD or d < bestD) then
				best, bestD = e, d
			end
		end
	end
	if best then
		best.amount = maxHatch(best)
		target, targetNote = best, ""
		return
	end
	if (cfg.filter or "") ~= "" and os.clock() >= regularOffUntil then
		target = { regular = true, id = cfg.filter, amount = maxHatch({}) }
		targetNote = "regular egg"
		return
	end
	target = nil
	targetNote = ("no egg matching '%s' within %d studs"):format(cfg.filter or "", cfg.range)
end

---------------------------------------------------------------- hatch engine
local MAX_IN_FLIGHT = 12
local MAX_LAUNCH_PER_FRAME = 4

local hatching = false
local status, statusKind = "Idle", "idle"
local inFlight = 0
local blockedUntil = 0 -- only for real problems (no coins / errors), never for the rate limit
local failStreak = 0
local amountCap = {}
local sessionHatched, acceptedReqs = 0, 0
local lastLaunchAt, lastSave, lastSuccessAt = 0, 0, 0

-- self-tuning pace: seconds between request launches
local interval = 0.02
local nextLaunchAt = 0
local successStreak, lastAdjustAt = 0, 0

local function setStatus(text, kind)
	status, statusKind = text, kind
end

local function addHatched(id, n)
	sessionHatched += n
	stats.total += n
	stats.perEgg[id] = (tonumber(stats.perEgg[id]) or 0) + n
	statsDirty = true
end

local RATE_WORDS = { "quickly", "too fast", "slow down", "cooldown", "rate limit", "too many", "spam" }
local function isRateLimit(text)
	for _, w in ipairs(RATE_WORDS) do
		if string.find(text, w, 1, true) then
			return true
		end
	end
	return false
end

-- the server turned a request down for being too early: widen the spacing a little (once per burst)
local function onRateLimited()
	successStreak = 0
	local now = os.clock()
	if now - lastAdjustAt < 0.15 then
		return
	end
	lastAdjustAt = now
	interval = math.min(2, math.max(0.02, interval * 1.3 + 0.004))
	nextLaunchAt = math.max(nextLaunchAt, now + interval)
end

-- a run of accepted requests: probe for a slightly faster pace
local function onAccepted()
	successStreak += 1
	if successStreak >= 12 then
		successStreak = 0
		interval = interval * 0.94
		if interval < 0.006 then
			interval = 0
		end
	end
end

local function requestHatch(egg, amount)
	if egg.regular then
		return netInvoke("Eggs_RequestPurchase", egg.id, amount)
	end
	return netInvoke("CustomEggs_Hatch", egg.uid, amount)
end

local function worker(egg, amount)
	local ok, res, msg = requestHatch(egg, amount)
	inFlight = math.max(0, inFlight - 1)
	if not alive then
		return
	end
	if ok and res == true then
		failStreak = 0
		acceptedReqs += 1
		lastSuccessAt = os.clock()
		addHatched(egg.id, amount)
		setStatus("Hatching", "ok")
		onAccepted()
		return
	end
	local text = string.lower(tostring(msg or res or ""))
	if isRateLimit(text) then
		onRateLimited()
	elseif string.find(text, "afford", 1, true) or string.find(text, "enough", 1, true) or string.find(text, "coin", 1, true) then
		if amount > 1 then
			amountCap[egg.id] = { n = math.max(1, math.floor(amount / 2)), at = os.clock() }
			setStatus(("Not enough coins for x%d, using less"):format(amount), "warn")
		else
			blockedUntil = os.clock() + 0.4
			setStatus("Not enough coins", "bad")
		end
	else
		failStreak += 1
		blockedUntil = os.clock() + 0.25
		setStatus("Hatch failed: " .. tostring(msg or res), "bad")
		if egg.regular and failStreak >= 4 then
			regularOffUntil = os.clock() + 15
			failStreak = 0
		end
	end
end

local function eggCost(egg)
	local d = egg.dir
	if type(d) == "table" and type(d.overrideCost) == "number" and d.currency ~= nil then
		return d.overrideCost, d.currency
	end
	return nil
end

local balanceAt, balanceVal, balanceCur = 0, nil, nil
local function coins(cur)
	if balanceCur ~= cur or os.clock() - balanceAt > 0.15 then
		balanceAt, balanceCur = os.clock(), cur
		local ok, have = gcall(G.CurrencyCmds.Get, cur)
		balanceVal = (ok and type(have) == "number") and have or nil
	end
	return balanceVal
end

local function heartbeatHatch()
	if not hatching then
		return
	end
	local now = os.clock()
	-- a request that never answers must not freeze hatching forever
	if inFlight > 0 and now - lastLaunchAt > 8 then
		inFlight = 0
	end
	if not modulesReady then
		setStatus("Loading game modules...", "warn")
		return
	end
	if now < blockedUntil then
		return
	end
	if now - targetAt > 0.3 or not target then
		refreshTarget()
	end
	local egg = target
	if not egg then
		setStatus("Waiting: " .. targetNote, "warn")
		return
	end
	local amount = egg.amount or 1
	local cap = amountCap[egg.id]
	if cap and now - cap.at < 10 then
		amount = math.min(amount, cap.n)
	end
	local cost, cur = eggCost(egg)
	if cost and cost > 0 and G.CurrencyCmds and type(G.CurrencyCmds.Get) == "function" then
		local have = coins(cur)
		if have then
			local affordable = math.floor(have / cost)
			if affordable < 1 then
				blockedUntil = now + 0.3
				setStatus("Not enough coins", "bad")
				return
			end
			amount = math.max(1, math.min(amount, affordable))
		end
	end
	if nextLaunchAt < now - 0.05 then
		nextLaunchAt = now - 0.05 -- never a big catch-up burst
	end
	local launched = 0
	while launched < MAX_LAUNCH_PER_FRAME and inFlight < MAX_IN_FLIGHT and nextLaunchAt <= now do
		inFlight += 1
		lastLaunchAt = now
		task.spawn(worker, egg, amount)
		nextLaunchAt += interval
		launched += 1
	end
end

track(RunService.Heartbeat:Connect(function()
	if alive then
		heartbeatHatch()
	end
end))

---------------------------------------------------------------- number display helpers
local function fmt(n)
	n = math.floor((tonumber(n) or 0) + 0.5)
	local s = tostring(n):reverse():gsub("(%d%d%d)", "%1,"):reverse()
	return (s:gsub("^,", ""))
end

-- 3 significant digits, so the big projection stays calm instead of flickering digits
local function sig3(n)
	if n < 1000 then
		return math.floor(n + 0.5)
	end
	local mag = 10 ^ (math.floor(math.log10(n)) - 2)
	return math.floor(n / mag + 0.5) * mag
end

-- glides a shown number toward its goal; snaps when it is close
local function approach(cur, goal, dt, speed)
	local diff = goal - cur
	if math.abs(diff) < 0.5 then
		return goal
	end
	return cur + diff * math.min(1, dt * speed)
end

local function countFor(filter)
	local f = string.lower(filter or "")
	local sum = 0
	for id, n in pairs(stats.perEgg) do
		if f == "" or string.find(string.lower(tostring(id)), f, 1, true) then
			sum += tonumber(n) or 0
		end
	end
	return sum
end

	local obj = Instance.new(class)

local lineOrder = 0
local function statLine(label)
	lineOrder += 1
	local row = new("Frame", { Size = UDim2.new(1, 0, 0, 17), BackgroundTransparency = 1, LayoutOrder = lineOrder }, card)
	local name = new("TextLabel", {
		BackgroundTransparency = 1, Size = UDim2.new(0.34, 0, 1, 0), Font = Enum.Font.Gotham, Text = label,
		TextColor3 = C.dim, TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left,
	}, row)
	local value = new("TextLabel", {
		BackgroundTransparency = 1, Position = UDim2.fromScale(0.34, 0), Size = UDim2.new(0.66, 0, 1, 0),
		Font = Enum.Font.GothamBold, Text = "-", TextColor3 = C.text, TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Right, TextTruncate = Enum.TextTruncate.AtEnd,
	}, row)
	return value, name
end

local KIND_COLOR = { ok = C.good, warn = C.warn, bad = C.bad, idle = C.dim }

local function setText(label, text)
	if label.Text ~= text then
		label.Text = text
	end
end

local sampleAcc, lastSessionCount, lastAccepted = 0, 0, 0

track(RunService.Heartbeat:Connect(function(dt)
	if not alive then
		return
	end

	-- once a second: smooth speed estimates (running average first, then a gentle moving average)
	sampleAcc += dt
	if sampleAcc >= 1 then
		local instRate = (sessionHatched - lastSessionCount) / sampleAcc
		local instReq = (acceptedReqs - lastAccepted) / sampleAcc
		lastSessionCount, lastAccepted, sampleAcc = sessionHatched, acceptedReqs, 0
		if hatching then
			warmSamples += 1
			local a = math.max(0.12, 1 / warmSamples)
			rateEMA += (instRate - rateEMA) * a
			reqEMA += (instReq - reqEMA) * a
		end
	end

	-- status: only claim "waiting" if the server has not accepted anything for a few seconds
	local kind = statusKind
	local text = status
	if not hatching then
		kind, text = "idle", "Idle"
	elseif kind == "ok" and os.clock() - lastSuccessAt > 3 and warmSamples > 0 then
		kind, text = "warn", "Waiting for the server..."
	end
	setText(vStatus, text)
	vStatus.TextColor3 = KIND_COLOR[kind] or C.text
	dot.BackgroundColor3 = hatching and C.good or C.off

	-- egg + live distance (recomputed every frame)
	if target then
		setText(vEgg, ("%s  x%d"):format(target.id, target.amount or 1))
		local root = hrp()
		if target.pos and root then
			local d = (target.pos - root.Position).Magnitude
			setText(vDist, ("%.1f studs"):format(d))
			vDist.TextColor3 = d <= cfg.range and C.good or C.warn
		else
			setText(vDist, target.regular and "regular egg" or "-")
			vDist.TextColor3 = C.dim
		end
	else
		setText(vEgg, "none in range")
		setText(vDist, "-")
		vDist.TextColor3 = C.dim
	end

	-- counters roll smoothly
	shown.session = approach(shown.session, sessionHatched, dt, 10)
	shown.total = approach(shown.total, stats.total, dt, 10)
	shown.count = approach(shown.count, countFor(cfg.filter), dt, 10)
	setText(vSession, fmt(shown.session))
	setText(vTotal, fmt(shown.total))
	setText(vCount, fmt(shown.count))
	setText(lCount, cfg.filter ~= "" and cfg.filter or "All eggs")

	-- speed + projection: smoothed, glides, 3 significant digits
	if warmSamples >= 3 or (not hatching and rateEMA > 0) then
		shown.hour = approach(shown.hour, rateEMA * 3600, dt, 1.5)
		shown.pace = approach(shown.pace, reqEMA, dt, 3)
		setText(vRate, ("%s / min"):format(fmt(sig3(shown.hour / 60))))
		setText(vProj, "~" .. fmt(sig3(shown.hour)))
		setText(paceLabel, ("Pace: %d req/s"):format(math.floor(shown.pace + 0.5)))
	else
		local msg = hatching and "measuring..." or "-"
		setText(vRate, msg)
		setText(vProj, msg)
		setText(paceLabel, "Pace: -")
	end
	paceLabel.TextColor3 = hatching and C.good or C.dim

	if statsDirty and os.clock() - lastSave > 15 then
		lastSave = os.clock()
		saveNow()
	end
end))

-- drag (direct, clamped to the screen)
local dragging, dragStart, startPos
bar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging, dragStart, startPos = true, input.Position, main.AbsolutePosition
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)
track(UIS.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		local screen, size = gui.AbsoluteSize, main.AbsoluteSize
		main.Position = UDim2.fromOffset(
			math.clamp(startPos.X + delta.X, 0, math.max(0, screen.X - size.X)),
			math.clamp(startPos.Y + delta.Y, 0, math.max(0, screen.Y - size.Y))
		)
	end
end))

---------------------------------------------------------------- unload
local function shutdown()
	if not alive then
		return
	end
	alive = false
	hatching = false
	for _, c in ipairs(conns) do
		pcall(function() c:Disconnect() end)
	end
	pcall(function()
		if animEnv and animOrig ~= nil then
			animEnv.PlayEggAnimation = animOrig
		end
	end)
	saveNow()
	gui:Destroy()
	genv.__EGGHATCHER = nil
end
closeBtn.MouseButton1Click:Connect(shutdown)
genv.__EGGHATCHER = shutdown

task.spawn(function()
	task.wait()
	relayout()
	main.Position = UDim2.new(0, 24, 0.5, -fullH / 2)
end)
