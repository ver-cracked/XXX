-- FPS Optimizer v3 | executor script
-- Tap the round bubble (or press RightShift) to open / close the window.
-- Drag the bubble or the window header to move them. F6 turns 3D back on if Pause 3D is active.
--
--   * every change is remembered and fully undone when a toggle is switched off or the window is closed
--   * background work shares one small per-frame time budget that shrinks by itself when the game lags
--   * memory cleanup is incremental and rate limited, never a forced full collect
--   * Beams and Highlights are never touched, so ESP / line features of other scripts keep working

local saveQueued = false
local snapshotStates -- filled in once the features exist

local function saveSoon()
	if not canFile or saveQueued or closed then
		return
	end
	saveQueued = true
	task.delay(0.6, function()
		saveQueued = false
		if snapshotStates then
			snapshotStates()
		end
		pcall(writefile, FILE, HttpService:JSONEncode(saved))
	end)
end

---------------------------------------------------------------- gui helpers
local function new(class, props, parent)
	local obj = Instance.new(class)
	for k, v in pairs(props) do
		pcall(function()
			obj[k] = v
		end)
	end
	obj.Parent = parent
	return obj
end

local function round(obj, r)
	return new("UICorner", {
		CornerRadius = typeof(r) == "UDim" and r or UDim.new(0, r),
	}, obj)
end

	return new("UIStroke", {
		Color = color,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		LineJoinMode = Enum.LineJoinMode.Round,
	}, obj)
end

	TweenService:Create(obj, TweenInfo.new(t, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

---------------------------------------------------------------- frame budget
-- All background work shares this small slice of every frame. If the game is
-- already struggling the slice shrinks, so the optimizer never piles on lag.
local frameDt = 1 / 60

local function budget()
	if frameDt > 1 / 25 then
		return 0.0012
	elseif frameDt > 1 / 45 then
		return 0.0025
	end
	return 0.004
end

---------------------------------------------------------------- state store
-- originals are kept per property in weak tables (one small entry per changed
-- object, and entries vanish by themselves when an object is destroyed)
local WEAK_K = { __mode = "k" }

local function getProp(inst, prop)
	return inst[prop]
end

local function putProp(inst, prop, value)
	inst[prop] = value
end

local function makeSet(def)
	return function(inst, prop, value)
		local ok, cur = pcall(getProp, inst, prop)
		if not ok or cur == value then
			return
		end
		local map = def.store[prop]
		if not map then
			map = setmetatable({}, WEAK_K)
			def.store[prop] = map
		end
		if map[inst] == nil then
			map[inst] = cur
		end
		pcall(putProp, inst, prop, value)
	end
end

-- puts everything back, spread over many frames so closing a feature never hitches
local function restoreAsync(store, done)
	task.spawn(function()
		local t0 = os.clock()
		for prop, map in pairs(store) do
			for inst, old in pairs(map) do
				pcall(putProp, inst, prop, old)
				map[inst] = nil
				if os.clock() - t0 > budget() then
					task.wait()
					t0 = os.clock()
				end
			end
			store[prop] = nil
		end
		if done then
			done()
		end
	end)
end

---------------------------------------------------------------- memory helpers
local function totalMb()
	local ok, mb = pcall(function()
		return Stats:GetTotalMemoryUsageMb()
	end)
	return ok and mb or 0
end

local function luaMb()
	local ok, kb = pcall(collectgarbage, "count")
	return ok and kb / 1024 or 0
end

-- one garbage-collection cycle done in tiny steps over many frames (no big stall).
-- returns false when the environment does not allow stepping.
local function gentleGC(maxSeconds)
	local started, slice = os.clock(), os.clock()
	for _ = 1, 800 do
		local ok, finished = pcall(collectgarbage, "step", 48)
		if not ok then
			return false
		end
		if finished then
			return true
		end
		if os.clock() - slice > 0.0015 then
			task.wait()
			slice = os.clock()
		end
		if os.clock() - started > (maxSeconds or 4) then
			break
		end
	end
	return true
end

local function feature(def)
	def.store = {}

end

-- other players' characters are hidden locally (only you see the change)
local hideConns = {}
local hidePlayers
local function hideInstance(inst)
	if not hidePlayers.active then
		return
	end
	if inst:IsA("BasePart") then
		hidePlayers.set(inst, "LocalTransparencyModifier", 1)
	elseif inst:IsA("Decal") then
		hidePlayers.set(inst, "Transparency", 1)
	elseif inst:IsA("ParticleEmitter") or inst:IsA("Trail") then
		hidePlayers.set(inst, "Enabled", false)
	end
end

local function hideCharacter(char)
	if not char or char == player.Character then
		return
	end
	for _, inst in ipairs(char:GetDescendants()) do
		hideInstance(inst)
	end
	table.insert(hideConns, char.DescendantAdded:Connect(hideInstance))
end

local function hookPlayer(p)
	if p == player then
		return
	end
	if p.Character then
		task.spawn(hideCharacter, p.Character)
	end
	table.insert(hideConns, p.CharacterAdded:Connect(function(char)
		task.defer(hideCharacter, char)
	end))
end

hidePlayers = feature({
	name = "Hide players",
	desc = "Hides other players (only for you)",
	usesStore = true,
	on = function()
		for _, p in ipairs(Players:GetPlayers()) do
			hookPlayer(p)
		end
		table.insert(hideConns, Players.PlayerAdded:Connect(hookPlayer))
	end,
	off = function()
		for _, c in ipairs(hideConns) do
			pcall(function()
				c:Disconnect()
			end)
		end
		table.clear(hideConns)
	end,
})

-- hides small anchored parts far from the camera. Only parts that really are
-- static decoration are tracked, and the work is spread over many frames.
local CULL_RANGE = 260
local cullList, cullSet, cullCursor = {}, {}, 1
local cullHidden = setmetatable({}, WEAK_K)

local cull
cull = feature({
	name = "Cull far objects",
	desc = "Hides small far away parts",
	handler = function(inst, set)
		if cullSet[inst] or not inst:IsA("BasePart") or inst:IsA("Terrain") then
			return
		end
		if inst.Anchored and inst.Transparency < 1 and inst.Size.Magnitude < 100 then
			local char = player.Character
			if not (char and inst:IsDescendantOf(char)) then
				cullSet[inst] = true
				cullList[#cullList + 1] = inst
			end
		end
	end,
	off = function()
		table.clear(cullList)
		table.clear(cullSet)
		table.clear(cullHidden)
		cullCursor = 1
	end,
})

local function pumpCull(deadline)
	if not cull.active then
		return
	end
	local cam = Workspace.CurrentCamera
	local n = #cullList
	if not cam or n == 0 then
		return
	end
	local origin = cam.CFrame.Position
	local hideSq = CULL_RANGE * CULL_RANGE
	local showSq = (CULL_RANGE * 0.9) * (CULL_RANGE * 0.9)
	local i, steps = cullCursor, 0
	local maxSteps = math.min(n, 3000)
	while steps < maxSteps and n > 0 do
		if i > n then
			i = 1
		end
		local part = cullList[i]
		if part.Parent == nil then
			-- destroyed: swap-remove so the list never grows stale
			cullSet[part] = nil
			cullHidden[part] = nil
			cullList[i] = cullList[n]
			cullList[n] = nil
			n -= 1
		else
			local d = part.Position - origin
			local distSq = d.X * d.X + d.Y * d.Y + d.Z * d.Z
			if cullHidden[part] then
				if distSq < showSq then
					cull.set(part, "LocalTransparencyModifier", 0)
					cullHidden[part] = nil
				end
			elseif distSq > hideSq then
				cull.set(part, "LocalTransparencyModifier", 1)
				cullHidden[part] = true
			end
			i += 1
		end
		steps += 1
		if steps % 64 == 0 and os.clock() > deadline then
			break
		end
	end
	cullCursor = i
end

---------------------------------------------------------------- auto memory guard
-- Keeps an eye on memory and acts gently:
--   1. when the Lua heap has grown a lot, one incremental garbage-collection cycle (rate limited)
--   2. when total memory gets close to the crash zone, turns on the next memory saver, one at a time
-- It never runs while the game is lagging, and it undoes its own changes when switched off.
local TOTAL_LIMIT_MB = TOUCH and 1500 or 3500
local memoryLadder = { effects, meshDetail, textures } -- least visual impact first
local autoEnabled = {}
local setFeature

local autoMem
autoMem = feature({
	name = "Auto memory",
	desc = "Cleans memory by itself, enables savers when low",
	on = function()
		task.spawn(function()
			local lastClean, lastStep = os.clock(), 0
			local baseLua = luaMb()
			while autoMem.active and not closed do
				task.wait(10)
				if not autoMem.active or closed then
					break
				end
				if frameDt < 1 / 20 then
					local now = os.clock()
					local lua = luaMb()
					if lua > 150 and lua > baseLua * 1.4 and now - lastClean > 90 then
						gentleGC(3)
						lastClean = now
						baseLua = luaMb()
					end
					if totalMb() > TOTAL_LIMIT_MB * 0.85 and now - lastStep > 40 then
						for _, def in ipairs(memoryLadder) do
							if not def.active and not def.restoring then
								setFeature(def, true)
								autoEnabled[def] = true
								lastStep = now
								notify("Auto: " .. def.name)
								break
							end
						end
					end
				end
			end
		end)
	end,
	off = function()
		for def in pairs(autoEnabled) do
			setFeature(def, false)
		end
		table.clear(autoEnabled)
	end,
})

function snapshotStates()
	for _, def in ipairs(features) do
		if def ~= pause3d then
			saved.states[def.name] = def.active and not autoEnabled[def] or false
		end
	end
end

---------------------------------------------------------------- engine
local roots = { Workspace, Lighting }

-- new instances are queued and handled inside the frame budget
local queue, qHead, qTail = {}, 1, 0
local QUEUE_MAX = 30000
local needRescan = false

local scan = { active = false, stack = {}, list = {} }

local function requestScan()
	table.clear(scan.list)
	for _, def in ipairs(handlerDefs) do
		if def.active then
			scan.list[#scan.list + 1] = def
		end
	end
	if #scan.list == 0 then
		scan.active = false
		return
	end
	table.clear(scan.stack)
	for i = #roots, 1, -1 do
		scan.stack[#scan.stack + 1] = roots[i]
	end
	scan.active = true
end

local function runHandlers(inst)
	for i = 1, #handlerDefs do
		local def = handlerDefs[i]
		if def.active then
			pcall(def.handler, inst, def.set)
		end
	end
end

local function pumpQueue(deadline)
	local n = 0
	while qHead <= qTail do
		local inst = queue[qHead]
		queue[qHead] = nil
		qHead += 1
		if inst and inst.Parent then
			runHandlers(inst)
		end
		n += 1
		if n % 32 == 0 and os.clock() > deadline then
			return
		end
	end
	qHead, qTail = 1, 0
end

local function pumpScan(deadline)
	if not scan.active then
		if needRescan and qHead > qTail then
			needRescan = false
			requestScan()
		end
		return
	end
	local stack, list = scan.stack, scan.list
	local n = 0
	while #stack > 0 do
		local inst = stack[#stack]
		stack[#stack] = nil
		for i = 1, #list do
			local def = list[i]
			if def.active then
				pcall(def.handler, inst, def.set)
			end
		end
		local ok, kids = pcall(inst.GetChildren, inst)
		if ok then
			for i = 1, #kids do
				stack[#stack + 1] = kids[i]
			end
		end
		n += 1
		if n % 24 == 0 and os.clock() > deadline then
			return
		end
	end
	scan.active = false
end

function setFeature(def, state)
	if def.active == state or def.restoring then
		return
	end
	def.active = state
	if def.handler then
		activeHandlers += state and 1 or -1
	end
	if state then
		if def.on then
			pcall(def.on)
		end
		if def.handler then
			requestScan()
		end
	else
		if def.off then
			pcall(def.off)
		end
		if def.handler or def.usesStore then
			local old = def.store
			def.store = {}
			def.restoring = true
			restoreAsync(old, function()
				def.restoring = false
			end)
		end
	end
	if def.visual then
		def.visual(state)
	end
	saveSoon()
end

for _, root in ipairs(roots) do
	table.insert(connections, root.DescendantAdded:Connect(function(inst)
		if activeHandlers <= 0 then
			return
		end
		if qTail - qHead >= QUEUE_MAX then
			needRescan = true -- queue full: one background re-sweep will catch the rest
			return
		end
		qTail += 1
		queue[qTail] = inst
	end))
end

local statFrames, statLast = 0, os.clock()
local statText, fpsNow = "", 60

table.insert(connections, RunService.Heartbeat:Connect(function(dt)
	frameDt = frameDt * 0.9 + dt * 0.1

	statFrames += 1
	local now = os.clock()
	if now - statLast >= 0.5 then
		fpsNow = math.floor(statFrames / (now - statLast) + 0.5)
		if hint ~= "" then
			statText = hint
		elseif now < noteUntil then
			statText = note
		else
			statText = ("%d FPS  |  %d MB  |  Lua %d MB"):format(fpsNow, math.floor(totalMb()), math.floor(luaMb()))
		end
		statFrames, statLast = 0, now
	end

	if activeHandlers <= 0 and not scan.active then
		return
	end
	local deadline = os.clock() + budget()
	pumpQueue(deadline)
	pumpScan(deadline)
	pumpCull(deadline)
end))

---------------------------------------------------------------- placement, drag, open / close
local function place(obj, x, y)
	local screen = gui.AbsoluteSize
	local size = obj.AbsoluteSize
	x = math.clamp(x, 0, math.max(0, screen.X - size.X))
	y = math.clamp(y, 0, math.max(0, screen.Y - size.Y))
	obj.Position = UDim2.fromScale(x / math.max(1, screen.X), y / math.max(1, screen.Y))
end

local function makeDraggable(handle, target, savedKey, onTap)
	local dragging, moved, startMouse, startPos = false, false, Vector2.zero, Vector2.zero
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging, moved = true, false
			startMouse = Vector2.new(input.Position.X, input.Position.Y)
			startPos = target.AbsolutePosition
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					if moved then
						local pos = target.Position
						saved[savedKey] = { pos.X.Scale, pos.Y.Scale }
						saveSoon()
					elseif onTap then
						onTap()
					end
				end
			end)
		end
	end)
	table.insert(connections, UIS.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = Vector2.new(input.Position.X, input.Position.Y) - startMouse
			if not moved and delta.Magnitude < 6 then
				return
			end
			moved = true
			place(target, startPos.X + delta.X, startPos.Y + delta.Y)
		end
	end))
end

local isOpen = false
local fitScale = 1
local openToken = 0

local function refit()
	local screen = gui.AbsoluteSize
	fitScale = math.min(1, (screen.X - 16) / WIN_W, (screen.Y - 16) / WIN_H)
	winScale.Scale = fitScale * (isOpen and 1 or 0.92)
end

local function setOpen(state)
	isOpen = state
	openToken += 1
	local token = openToken
	saved.open = state
	saveSoon()
	if state then
		main.Visible = true
		winScale.Scale = fitScale * 0.92
		tween(winScale, 0.16, { Scale = fitScale })
		task.defer(function()
			place(main, main.AbsolutePosition.X, main.AbsolutePosition.Y)
		end)
	else
		tween(winScale, 0.12, { Scale = fitScale * 0.92 })
		task.delay(0.13, function()
			if openToken == token then
				main.Visible = false
			end
		end)
	end
	tween(bubbleStroke, 0.15, { Transparency = state and 0.55 or 0.1 })
end

makeDraggable(bar, main, "window")
makeDraggable(bubble, bubble, "bubble", function()
	setOpen(not isOpen)
end)

hideBtn.MouseButton1Click:Connect(function()
	setOpen(false)
end)

	end)
end))

---------------------------------------------------------------- close / keys
local function shutdown()
	if closed then
		return
	end
	if snapshotStates then
		snapshotStates()
	end
	if canFile then
		pcall(writefile, FILE, HttpService:JSONEncode(saved))
	end
	closed = true
	for _, def in ipairs(features) do
		if def.active then
			setFeature(def, false)
		end
	end
	pcall(function()
		RunService:Set3dRenderingEnabled(true)
	end)
	for _, c in ipairs(connections) do
		pcall(function()
			c:Disconnect()
		end)
	end
	gui:Destroy()
	if getgenv then
		getgenv().__FPSOPT = nil
	end
end

closeBtn.MouseButton1Click:Connect(shutdown)
if getgenv then
	getgenv().__FPSOPT = shutdown
end

table.insert(connections, UIS.InputBegan:Connect(function(input, processed)
	if processed then
		return
	end
	if input.KeyCode == Enum.KeyCode.RightShift then
		setOpen(not isOpen)
	elseif input.KeyCode == Enum.KeyCode.F6 and pause3d.active then
		setFeature(pause3d, false)
	end
end))
